## 12. The tracker artifact and how to recreate it

### What exists
- Published page (private to the user): https://claude.ai/artifact/EWChMpRK4RLcVgdhfFrvcu, "Danadin's Road to 60", version 11, capabilities `db` and `user`.
- Repo: `Dandotmd/BiS-hunter`, branch **`claude/wonderful-curie-mw0z1j`**, folder `tracker/` containing `index.html`, `data.js`, `README.md`. (The older TBC "BiSHunter" Flask app in the repo root is unrelated and untouched.)
- The page has: header with level stepper and profession boxes; **Focus now** (recomputed next run, pickups grouped by hub, craftable upgrades, weakest slots, milestones); **Dungeon route** (expandable cards with quest checklists, prerequisites, copyable `/way` coordinates, loot); **Gear targets**; **Milestones**; **Tell Claude** (open questions).
- `data.js` holds the plan. Edit it to change the plan; `index.html` renders it and reads saved progress.

### Database (the `db` capability), collection `tracker`
Two documents.

`tracker/character` (as of 2026-11-19, version 6):
```json
{"level": 42, "mining": 194, "blacksmithing": 173, "firstAid": 150}
```
Optional field `gold` (number). A field set to `""` or deleted falls back to the baseline in `data.js`.

`tracker/progress` (as of 2026-11-19, version 8):
```json
{
  "d_gnomer": "done", "d_rfk": "done", "d_smgy": "done", "d_smlib": "done",
  "d_dalaran": "done", "d_rfd": "done", "d_drowned": "done",
  "d_smarm": "done", "d_smcath": "done",
  "q_1048": "have",
  "q_3341": "done", "q_6521": "done", "q_97196": "done",
  "q_6626": "done", "q_3523": "done", "q_3525": "done",
  "g_abomination-chain": "1", "g_vanquisher-s-sword": "1", "g_aegis-of-the-scarlet-commander": "1",
  "m_dualspec": "1", "m_mount": "1"
}
```
Key conventions: `d_<dungeon id>` = `"done"`; `q_<quest id>` = `"have"` (in log) or `"done"`; `g_<slug>` = `"1"` when a gear target is obtained, where slug is the item name lowercased with every non-alphanumeric run turned into `-`; `m_<milestone id>` = `"1"`. Setting a value to `""` clears it. Suggested updates not yet recorded: `q_1048` should become `"done"` once the user confirms the turn-in; Krol'dok chain progress has no keys yet (the dungeon quest ids are `q_94975`, `q_94968`, `q_94972`, `q_94973`).

2. Publish `index.html` as an artifact with `data.js` as a supporting file, icon `shield`, title `Danadin's Road to 60`, capabilities `{"db": {}, "user": {}}`.
3. Seed the two documents above through the artifact database tool (`tracker/character` and `tracker/progress`).
4. Update `data.js` and republish whenever the plan or facts change. Update the two documents when progress changes. Mirror each change in both places (the page seeds some progress from `data.js`, e.g. completed dungeons).

      b: "Several best-in-slot tank pieces are crafted. Check your trainer for the recipes; Mining " + S.char.mining + " covers the ore.",
      l: can.map(function (c) { return "<b>" + esc(c.t.name) + "</b> (" + c.g.slot + ", skill " + c.t.bs + ") — " + esc(c.t.stats); })
        .concat(nxt.slice(0, 3).map(function (c) { return '<span class="muted">At ' + c.t.bs + ": " + esc(c.t.name) + " (" + c.g.slot + (c.t.at > L ? ", level " + c.t.at : "") + ")</span>"; })) });

    /* weakest slots with something obtainable now */
    var weak = T.gear.filter(function (g) { return g.ilvl < L - 6; }).sort(function (a, b) { return a.ilvl - b.ilvl; }).slice(0, 4).map(function (g) {
      var t = g.targets.filter(function (t) { return !S.prog["g_" + slug(t.name)] && tReady(t); })[0];
      return t ? "<b>" + g.slot + "</b> (" + (g.ilvl ? "item level " + g.ilvl : "empty") + ") → " + esc(t.name) + ' <span class="muted">' + esc(srcText(t)) + "</span>" : null;
    }).filter(Boolean);
    if (weak.length) out.push({ k: "Gear", t: "Weakest slots you can fix now", b: "", l: weak });

    /* milestones */
    var ms = T.milestones.filter(function (m) { return !S.prog["m_" + m.id] && m.at - L <= 4; });
    if (ms.length) out.push({ k: "Milestones", t: L < 40 ? (40 - L) + " level" + (40 - L > 1 ? "s" : "") + " to 40" : "Ready now", b: "",
      l: ms.map(function (m) { return "<b>" + esc(m.name) + "</b>" + (m.at > L ? ' <span class="muted">at ' + m.at + "</span>" : "") + " — " + esc(m.detail); }) });

    $("focus-sub").textContent = "Recomputed for level " + L;
    $("focus").innerHTML = out.map(function (o) {
      return '<li><span class="k">' + esc(o.k) + '</span><span class="t">' + esc(o.t) + "</span>" + (o.b ? "<span>" + esc(o.b) + "</span>" : "") + (o.l.length ? "<ul>" + o.l.map(function (x) { return "<li>" + x + "</li>"; }).join("") + "</ul>" : "") + "</li>";
    }).join("") || '<li><span class="t">Nothing queued.</span></li>';
  }

  function srcText(t) {
    if (t.how === "craft") return "craft, Blacksmithing " + t.bs;
    var d = t.from ? dById(t.from).name : "";
    return (t.how === "quest" ? "quest: " : t.how === "buy" ? "buy: " : "") + [t.src, d].filter(Boolean).join(", ");
  }

  var openSet = null;
  function renderDungeons() {
    var box = $("dungeons");
    if (openSet === null) { openSet = {}; var f = T.dungeons.filter(function (d) { return dStatus(d) === "now"; })[0]; if (f) openSet[f.id] = 1; }
    else { openSet = {}; [].forEach.call(box.querySelectorAll("details[open]"), function (e) { openSet[e.dataset.id] = 1; }); }
    box.innerHTML = T.dungeons.map(function (d) {
      var st = dStatus(d), qs = d.quests.filter(function (q) { return !q.deferred; }), got = qs.filter(function (q) { return qState(q); }).length;
      var body = "";
      if (d.note) body += "<p>" + esc(d.note) + "</p>";
      if (d.quests.length) body += '<div><div class="lab">Quests</div></div><ul class="qs">' + d.quests.map(function (q) {
        var s = qState(q);
        return '<li class="q" data-s="' + s + '"><button class="qb" data-q="' + esc(q.id) + '" data-s="' + s + '" aria-label="' + esc(q.name) + ": " + (s || "need") + '">' + (q.deferred ? "later" : s === "have" ? "in log" : s === "done" ? "done" : "need") + "</button>" +
          '<div class="qt"><span class="qn">' + (typeof q.id === "number" ? '<a href="https://www.wowhead.com/forever/quest=' + q.id + '" target="_blank" rel="noopener">' + esc(q.name) + "</a>" : esc(q.name)) +
          (q.isNew ? ' <span class="new">new</span>' : "") + (q.lvl ? ' <span class="muted small">min ' + q.lvl + "</span>" : "") + "</span>" +
          '<span class="small"><span class="muted">' + esc(q.giver) + " · " + esc(q.zone) + (q.area ? ", " + esc(q.area) : "") + "</span>" + wayBtn(q) + "</span>" +
          (q.pre ? '<span class="pre">Prerequisite: ' + esc(q.pre) + "</span>" : "") +
          "<span>" + esc(q.task) + "</span>" + (q.reward ? '<span class="rew">Reward: ' + esc(q.reward) + "</span>" : "") + "</div></li>";
      }).join("") + "</ul>";
      if (d.loot && d.loot.length) body += '<div><div class="lab">Worth it for a tank</div><ul class="loot">' + d.loot.map(function (l) { return "<li>" + esc(l) + "</li>"; }).join("") + "</ul></div>";
      if (d.guide) body += '<a class="small" href="' + d.guide + '" target="_blank" rel="noopener">Wowhead guide</a>';
      body += '<button class="btn" data-d="' + d.id + '">' + (st === "done" ? "Mark as not run" : "Mark dungeon done") + "</button>";
      return '<details class="d" data-id="' + d.id + '" data-st="' + st + '"' + (openSet[d.id] ? " open" : "") + "><summary>" +
        '<span class="dn">' + esc(d.name) + (d.isNew ? ' <span class="new">new</span>' : "") + '</span><span class="pill p-' + st + '">' + LABEL[st] + "</span>" +
        '<span class="dm"><span class="r">' + d.range[0] + "–" + d.range[1] + "</span><span>" + esc(d.zone) + "</span>" + (qs.length && st !== "done" ? "<span>" + got + "/" + qs.length + " quests picked up</span>" : "") + "</span></summary>" +
        '<div class="dbody">' + body + "</div></details>";
    }).join("");
  }

  function renderGear() {
    var L = S.char.level;
    $("gear").innerHTML = T.gear.map(function (g) {
      var weak = g.ilvl < L - 6;
      return '<div class="g"><div><div class="gs">' + g.slot + '</div><div class="gc">' + esc(g.cur) + (g.ilvl ? " · " + g.ilvl : "") + "</div>" + (weak ? '<div class="weak">' + (g.ilvl ? (L - g.ilvl) + " levels behind" : "empty") + "</div>" : "") + "</div>" +
        '<div class="gt">' + g.targets.map(function (t) {
          var k = "g_" + slug(t.name), got = !!S.prog[k], rdy = tReady(t);
          return '<div class="t1' + (got ? " got" : "") + '"><input type="checkbox" id="' + k + '" data-g="' + k + '"' + (got ? " checked" : "") + '><label for="' + k + '"><b>' + esc(t.name) + "</b>" + (t.stats ? " — " + esc(t.stats) : "") + ' <span class="muted small">' + esc(srcText(t)) + "</span></label>" +
            (got ? "" : '<span class="tag' + (rdy ? " go" : "") + '">' + (rdy ? "get now" : t.at > L ? "level " + t.at : t.how === "craft" ? "skill " + t.bs : "soon") + "</span>") + "</div>";
        }).join("") + "</div></div>";
    }).join("");
  }

  function renderRest() {
    $("ms").innerHTML = T.milestones.slice().sort(function (a, b) { return a.at - b.at; }).map(function (m) {
      var k = "m_" + m.id;
      return '<li><input type="checkbox" id="' + k + '" data-g="' + k + '"' + (S.prog[k] ? " checked" : "") + '><label for="' + k + '"><b>' + esc(m.name) + '</b> <span class="muted small">level ' + m.at + "</span><br><span class=\"small\">" + esc(m.detail) + "</span></label></li>";
    }).join("");
    $("unk").innerHTML = T.unknowns.map(function (u) { return "<li>" + esc(u) + "</li>"; }).join("");
    $("foot").innerHTML = 'Game data checked ' + esc(T.checked) + ' against Wowhead, ForeverDB and Forever Codex. New-dungeon loot sources and anything past level 45 lean on Classic data and may differ in Forever. Gear snapshot from the <a href="' + C.armory + '" target="_blank" rel="noopener">armory</a>.';
  }

  function render() { renderHead(); renderFocus(); renderDungeons(); renderGear(); renderRest(); }

  /* ---------- events ---------- */
  document.addEventListener("click", function (e) {
    var b = e.target.closest("button"); if (!b) return;
    if (b.id === "lvl-up" && S.char.level < 60) return setC("level", S.char.level + 1);
    if (b.id === "lvl-down" && S.char.level > 1) return setC("level", S.char.level - 1);
    if (b.dataset.q) { e.preventDefault(); var k = "q_" + b.dataset.q, s = S.prog[k] || ""; return setP(k, s === "" ? "have" : s === "have" ? "done" : ""); }
    if (b.dataset.d) { var dk = "d_" + b.dataset.d; return setP(dk, S.prog[dk] === "done" ? "no" : "done"); }
    if (b.dataset.way) {
      var txt = b.dataset.way, old = b.textContent;
      var ok = function () { b.textContent = "copied"; setTimeout(function () { b.textContent = old; }, 900); };
      try { navigator.clipboard.writeText(txt).then(ok, function () {}); } catch (err) {}
    }
  });
  document.addEventListener("change", function (e) {
    var t = e.target;
    if (t.dataset.g) return setP(t.dataset.g, t.checked ? "1" : "");
    if (t.dataset.pf) { var v = Math.max(1, Math.min(300, parseInt(t.value, 10) || 1)); return setC(t.dataset.pf, v); }
  });

  /* ---------- boot ---------- */
  lsLoad(); render();
  function clean(o) { var r = {}; Object.keys(o || {}).forEach(function (k) { if (o[k] !== "" && o[k] != null) r[k] = o[k]; }); return r; }
  if (window.claude && window.claude.use) {
    window.claude.use("db").then(function (d) {
      if (!d) { say("Saved on this device only"); return; }
      db = d; say("");
      db.doc("tracker/character").onSnapshot(function (s) { if (s.exists) { Object.assign(S.char, clean(s.data())); render(); } }, function () {});
      db.doc("tracker/progress").onSnapshot(function (s) {
        if (!s.exists) return;
        var data = s.data() || {}, keep = {};
        /* seeded defaults stay unless the store says otherwise; "" or "no" clears */
        Object.keys(S.prog).forEach(function (k) { if (!(k in data)) keep[k] = S.prog[k]; });
        S.prog = Object.assign(keep, clean(data));
        render();
      }, function () {});
    });
  }
})();
